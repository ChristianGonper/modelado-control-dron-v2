\section*{Abreviaturas y acrónimos}
\addcontentsline{toc}{section}{Abreviaturas y acrónimos}

\hypersetup{allcolors=black}
\setlength{\parindent}{0cm}

\begin{description}[leftmargin=2.5cm, labelindent=0cm]
  \item[6DOF] Seis grados de libertad.
  \item[ALVINN] Vehículo terrestre autónomo en una red neuronal (\textit{Autonomous Land Vehicle in a Neural Network}).
  \item[BPTT] Retropropagación a través del tiempo (\textit{Backpropagation Through Time}).
  \item[CCW] Sentido de giro antihorario (\textit{Counter-Clockwise}).
  \item[CM] Centro de masas.
  \item[CPU] Unidad central de procesamiento (\textit{Central Processing Unit}).
  \item[CSV] Valores separados por comas (\textit{Comma-Separated Values}).
  \item[CW] Sentido de giro horario (\textit{Clockwise}).
  \item[DAgger] Agregación de conjuntos de datos (\textit{Dataset Aggregation}).
  \item[ENU] Sistema de referencia local East--North--Up.
  \item[FIFO] Primero en entrar, primero en salir (\textit{First In, First Out}).
  \item[FRD] Sistema de referencia de cuerpo Forward--Right--Down.
  \item[GNSS] Sistema global de navegación por satélite (\textit{Global Navigation Satellite System}).
  \item[GPU] Unidad de procesamiento gráfico (\textit{Graphics Processing Unit}).
  \item[GRU] Unidad recurrente con compuertas (\textit{Gated Recurrent Unit}).
  \item[ID] Dentro de la distribución de entrenamiento (\textit{in-distribution}).
  \item[IMU] Unidad de medida inercial (\textit{Inertial Measurement Unit}).
  \item[JSON] Notación de objetos de JavaScript (\textit{JavaScript Object Notation}).
  \item[LSTM] Memoria a corto y largo plazo (\textit{Long Short-Term Memory}).
  \item[MAE] Error absoluto medio (\textit{Mean Absolute Error}).
  \item[MLP] Perceptrón multicapa (\textit{Multilayer Perceptron}).
  \item[MSE] Error cuadrático medio (\textit{Mean Squared Error}).
  \item[NaN] No es un número (\textit{Not a Number}).
  \item[OOD] Fuera de la distribución de entrenamiento (\textit{out-of-distribution}).
  \item[PD] Controlador proporcional--derivativo.
  \item[PID] Controlador proporcional--integral--derivativo.
  \item[ReLU] Unidad lineal rectificada (\textit{Rectified Linear Unit}).
  \item[RK4] Método de Runge--Kutta de cuarto orden.
  \item[RMS] Raíz del valor cuadrático medio (\textit{Root Mean Square}).
  \item[RMSE] Raíz del error cuadrático medio (\textit{Root Mean Square Error}).
  \item[RNN] Red neuronal recurrente (\textit{Recurrent Neural Network}).
  \item[TOML] Formato de configuración minimalista (\textit{Tom's Obvious Minimal Language}).
  \item[UAV] Vehículo aéreo no tripulado (\textit{Unmanned Aerial Vehicle}).
  \item[YAML] Formato de serialización de datos (\textit{YAML Ain't Markup Language}).
  \item[ZOH] Bloqueador de orden cero (\textit{Zero-Order Hold}).
\end{description}
